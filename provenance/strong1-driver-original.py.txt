"""Reproduce the local, unpromoted strong-scale1 candidate from tactical ATD."""
from pathlib import Path
import argparse
import csv
import datetime as dt
import json
import os
import sys
import tempfile
import time

ROOT = Path(__file__).resolve().parent
# PEP420 scripts/vendor namespaces must never merge with a research checkout.
sys.path[:] = [str(ROOT)] + [p for p in sys.path if Path(p or ".").resolve() != ROOT and not any(
    (Path(p or ".").resolve()/name).exists() for name in
    ("second_zero", "scripts/wqq_campaign_raw_replay_v2.py", "vendor/mtah"))]
os.environ.setdefault("OMP_NUM_THREADS", "1")
os.environ.setdefault("OPENBLAS_NUM_THREADS", "1")
import reproduce as base
ALLOWED_OUTPUTS = set()


def reference_guard(event, args):
    if event != "open" or not isinstance(args[0], (str, bytes, os.PathLike)):
        return
    path = Path(os.fsdecode(args[0])).resolve()
    if path in ALLOWED_OUTPUTS:
        return
    mode, flags = args[1:3]
    if ((isinstance(mode, str) and any(c in mode for c in "wax+")) or
            isinstance(flags, int) and flags & (os.O_WRONLY | os.O_RDWR | os.O_CREAT)):
        return
    text = str(path).lower()
    if any(s in text for s in (".truth", "possession_gt", "/evaluation/gt/", ".actor")):
        raise PermissionError("Human/reference labels are excluded from inference")
    if path.suffix.lower() in {".csv", ".npz", ".npy"}:
        raise PermissionError("CSV and feature/prediction array caches are never inference inputs")
    if path.suffix.lower() in {".pt", ".pkl"} and not path.is_relative_to(ROOT):
        raise PermissionError("Only the pinned packaged original weights may be loaded")


def write(path, value):
    with path.open("x") as stream:
        json.dump(value, stream, indent=2, sort_keys=True, allow_nan=False)
        stream.write("\n")


def module_paths():
    result = {}
    for name, module in sys.modules.copy().items():
        if name in {"reproduce", "reproduce_strong_scale1"} or any(
                name == p or name.startswith(p + ".") for p in ("scripts", "second_zero", "vendor")):
            origin = getattr(module, "__file__", None)
            if origin is not None:
                if not Path(origin).resolve().is_relative_to(ROOT):
                    raise ValueError("Inference imported an unpackaged module: " + name)
                result[name] = str(Path(origin).resolve().relative_to(ROOT))
            else:
                paths = [Path(p).resolve() for p in getattr(module, "__path__", [])]
                if not paths or any(not p.is_relative_to(ROOT) for p in paths):
                    raise ValueError("Inference namespace has an unpackaged fallback: " + name)
                result[name] = dict(namespace_paths=[str(p.relative_to(ROOT)) for p in paths])
    if any("source_nn_candidate" in n or "player_motion_owner" in n for n in result):
        raise ValueError("The additional source-NN branch is excluded")
    return result


def reproduce(tactical_dir, output):
    sys.addaudithook(reference_guard)
    config = json.loads((ROOT / "strong1.json").read_text())
    original = json.loads((ROOT / "inference.json").read_text())
    base.verify(ROOT / "inference.json", config["base_inference_sha256"])
    base.verify_environment(original["runtime"])
    for name, digest in config["preserved_base_files"].items():
        base.verify(ROOT / name, digest)
    for name, digest in config["additional_files"].items():
        base.verify(ROOT / name, digest)
    if len(original["files"]) != 152 or any(
            config["preserved_base_files"].get(n) != h for n, h in original["files"].items()):
        raise ValueError("All 152 pinned original assets must remain exact")
    if config["status"] != "local_unpromoted_research_candidate" or config["source_NN_used"]:
        raise ValueError("Unexpected candidate role or source-NN dependency")
    from scripts import wqq_campaign_transition_replay_v1 as replay
    from second_zero import strong_scale1_candidate as candidate
    from second_zero.evaluate import validate_submission
    from second_zero.ingest import load_metadata, load_tracking
    from vendor.mtah import games
    if candidate.POLICY != config["policy"] or original["policy"] != replay.POLICY:
        raise ValueError("The fixed composition changed")
    np, chain = replay.np, replay.chain
    if chain.ROOT != ROOT or base.ROOT != ROOT:
        raise ValueError("Wrong package root")
    tracking = tactical_dir / "BAR-ATH_tactical_atd.txt"
    metadata = tactical_dir / "BAR-ATH_tactical_atd.xml"
    base.verify(tracking, original["inputs"]["tracking"])
    base.verify(metadata, original["inputs"]["metadata"])
    began = time.monotonic()
    game = games.GAMES[games.GAME]
    count, fps = game["n_video_frames"], game["fps"]
    meta = load_metadata(metadata)
    if (count, fps) != (199319, 25) or (meta["n_frames"], meta["fps"]) != (count, fps):
        raise ValueError("Unexpected official timeline")
    track = load_tracking(tracking, meta, count)
    pitch = np.array(game["pitch"], np.float64)
    team = np.array([0 if s == "H" else 1 if s == "A" else -1 for s in track.player_team], np.int64)
    keep = team >= 0
    predictions = np.full(count, 3, np.int8)
    rows = []
    base.build_kernels(chain, original["spec"])
    chain.torch.set_num_threads(1)
    with chain.threadpool_limits(1):
        bundle = chain.load_bundle(original["spec"])
        for half, start, last in games.half_video_blocks(games.GAME):
            end = last + 1
            raw = dict(ball=(track.ball_xy[start:end].astype(np.float64) * pitch).astype(np.float32),
                       players=(track.players_xy[start:end, keep].astype(np.float64) * pitch).astype(np.float32),
                       team=team[keep], pitch=pitch)
            arrays, _ = candidate.predict_raw(raw, bundle)
            labels = arrays["predicted"]
            if labels.shape != (end - start,) or not np.isin(labels, (0, 1, 2, 3)).all():
                raise ValueError("Invalid complete-half output")
            predictions[start:end] = labels
            rows.append(dict(half=int(half), start=start, stop=end, frames=end-start,
                             raw_identity=chain.raw_identity(raw)))
            print(json.dumps(dict(half=int(half), frames=end-start,
                                  elapsed_seconds=time.monotonic()-began)), flush=True)
            del raw, arrays
    imports = module_paths()
    output.parent.mkdir(parents=True, exist_ok=True)
    with tempfile.TemporaryDirectory(prefix=".strong1-", dir=output.parent) as temp:
        staged = Path(temp) / "predictions.csv"
        with staged.open("x", encoding="utf-8", newline="") as stream:
            writer = csv.writer(stream)
            writer.writerow(("frame", "label"))
            writer.writerows((i, "HADX"[int(v)]) for i, v in enumerate(predictions))
        # Validators read only the newly created output, never a prediction input.
        # This explicit, scoped exception leaves all other CSV/cache reads blocked.
        ALLOWED_OUTPUTS.add(staged.resolve())
        try:
            validation = validate_submission(staged, count)
            digest = base.sha256(staged)
        finally:
            ALLOWED_OUTPUTS.remove(staged.resolve())
        if digest != config["output_sha256"]:
            raise ValueError("Fresh strong-scale1 output differs from the pinned production identity")
        os.link(staged, output)
    result = dict(candidate_sha256=base.sha256(ROOT / "strong1.json"), output_sha256=digest,
                  rows=count, half_inputs=rows, format_validation=validation, source_NN_used=False,
                  reference_reads=[], external_cache_inputs=[], module_paths=imports,
                  elapsed_seconds=time.monotonic()-began, created_utc=dt.datetime.now(dt.timezone.utc).isoformat(),
                  hidden_accuracy_unknown=True, goal_075_demonstrated=False,
                  promoted=False, submitted=False, published=False)
    write(output.with_suffix(output.suffix + ".receipt.json"), result)
    print(json.dumps(dict(output=str(output), sha256=digest, rows=count,
                          elapsed_seconds=result["elapsed_seconds"])), flush=True)


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--tactical-dir", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    if args.output.exists() or args.output.with_suffix(args.output.suffix + ".receipt.json").exists():
        parser.error("Use new output and receipt paths")
    reproduce(args.tactical_dir, args.output)
